script.jsANIMATION LOOP / 003
function animateStars() {
  ctx.clearRect(0, 0, width, height);
  updateCosmicPalette();
  drawConstellations();

  stars.forEach(star => {
    star.twinklePhase += star.twinkleSpeed;
    // update glow and movement
    star.y -= star.speed;
  });

  requestAnimationFrame(animateStars);
}
